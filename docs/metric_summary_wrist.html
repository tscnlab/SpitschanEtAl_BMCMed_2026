Wrist metric-error summary workbook

Authors

Johannes Zauner

Manuel Spitschan

Purpose

This workbook is a reproducible metric-based analysis used for the wrist-versus-glasses comparison of the MeLiDos multi-country personal light exposure datasets. It starts from the prepared participant-day metric file (data/prepared_metrics.RData), fits the metric models, and builds the summary table object metrics_summary.

Methods

Dataset and reference position

The analysis uses the MeLiDos field-study wearable light-exposure data. The study collected synchronous light measurements from multiple wearing positions. For the metric-based analysis, glasses-mounted measurements are treated as the reference proxy for near-eye exposure, and wrist-worn measurements are compared against that reference.

Prepared metric data

The prepared data used here are created in notebooks/JZ_metric_preparation.qmd. In brief, the preparation workflow:

  1. imports glasses and wrist light logger data from melidosData, retaining melanopic equivalent daylight illuminance (MEDI) and photopic illuminance (LIGHT);
  2. removes the MPI site from this analysis because comparable ActLumus chest/wrist data were not collected there;
  3. aligns both wearing positions into one time series and keeps only timestamps where both positions have valid melanopic measurements;
  4. merges sleep-diary windows and non-wear logs;
  5. removes non-wear intervals unless they are sleep intervals;
  6. removes implausible high values (>= 100000 lx) for melanopic and photopic measurements;
  7. removes hours with less than 50% data availability and days with less than 80% availability;
  8. calculates daily light-exposure metrics per participant, date, and wearing position, plus participant-level rhythm metrics; and
  9. stores the resulting long metric table as data/prepared_metrics.RData.

Metric modelling

Each metric is modelled according to data/metric_types.xlsx, which defines the metric family, model engine, response transformation, and random-effects structure. where position has glasses as the reference level and wrist as the comparison level. Gaussian, Tweedie, or Poisson families are used according to the metric specification. If a mixed model does not converge or is singular, the random-effects structure is simplified stepwise, following the original metric-comparison notebook.

Summary statistic

For each metric, the wrist error is the estimated wrist-minus-glasses contrast from the fitted model, divided by the absolute estimated glasses reference value. The resulting relative error is summarized by metric type as:

  • median absolute wrist error;
  • minimum and maximum absolute wrist error;
  • number of significant wrist errors (p <= 0.05) out of all estimable metrics; and
  • count of positive and negative wrist deviations.

Two overall rows are added: one across all estimable metrics and one restricted to metrics with significant wrist errors.

metrics_nested <-
  metrics_nested |>

Formula_cols <- names(metrics_nested) |> keep(\(x) str_detect(x, "H1|H0_"))

make_formula <- function(response, rhs) {
  if (is.na(rhs) || is.na(response)) {
    return(NULL)
  }

  as.formula(paste(response, rhs))
}

metrics_nested <-
  metrics_nested |>
  mutate(
    response = str_replace(response, "metric", "value"),
    across(all_of(Formula_cols), \(x) map2(x, response, \(rhs, resp) make_formula(resp, rhs)))
  )
main_formula <- value ~ site * position + site + position + (position | Id) + (1 | Date)
red_formula <- value ~ site * position + site + position + (position | Id)
red_formula2 <- value ~ site * position + site + position + (1 | Id)

rhs_same <- function(f1, f2) {
  identical(
    attr(terms(formula(f1, rhs = 1)), "term.labels"),
    attr(terms(formula(f2, rhs = 1)), "term.labels")
  )
}

fit_model <- function(data, formula, engine, family) {
  if (is.null(formula)) {
    return(NULL)
  }

  model <- switch(
    engine,
    "lm" = lm(formula, data = data, contrasts = list(site = "contr.sum")),
    "glm" = glm(formula, data = data, family = family, contrasts = list(site = "contr.sum")),
    "lmer" = lmer(formula, data = data, contrasts = list(site = "contr.sum")),
    "glmer" = glmer(formula, data = data, family = family, contrasts = list(site = "contr.sum")),
    "glmmTMB" = glmmTMB(formula, data = data, family = family, contrasts = list(site = "contr.sum"))
  )

  did_converge <- performance::check_convergence(model) && !performance::check_singularity(model)

  if (!did_converge) {
    if (rhs_same(main_formula, formula)) {
      formula <- update(formula, . ~ . - (1 | Date))
    } else if (rhs_same(red_formula, formula)) {
      formula <- update(formula, . ~ . - (position | Id) + (1 | Id))
    } else if (rhs_same(red_formula2, formula)) {
      formula <- update(formula, . ~ . - site:position)
    } else {
      return(model)
    }

    model <- fit_model(data, formula, engine, family)
  }

  model
}

Extract wrist errors

metric_results <-
  metrics_nested |>
  mutate(
    main_effect = map_dbl(
      H1_model,
      \(model) model |>
        emmeans(specs = ~1, at = list(position = "glasses")) |>
        as_tibble() |>
        pull(2)
    ),
    position_effect = map(
      H1_model,
      \(model) model |>
        emmeans(specs = ~position) |>
        contrast(method = "trt.vs.ctrl") |>
        as_tibble()
    ),
    wrist_error = map_dbl(position_effect, \(x) x$estimate[x$contrast == "wrist - glasses"]),
    p_wrist = map_dbl(position_effect, \(x) x$p.value[x$contrast == "wrist - glasses"]),
    wrist_error = wrist_error / abs(main_effect)
  ) |>
  select(metric, metric_family, metric_type, wrist_error, p_wrist)
format_metric <- function(x) {
  case_when(
    x %in% c("MDER", "nvRD") ~ x,
    x == "centroidLE" ~ "Centroid LE",
    x == "midpointCE" ~ "Midpoint CE",
    x == "mean_MEDI" ~ "Mean mel EDI",
    TRUE ~ x |>
      str_replace_all("_", " ") |>
      str_to_sentence() |>
      str_replace_all("(\\d+)(?!\\d|h)", "\\1 lx mel EDI")
  )
}

metric_results <-

Build metrics_summary

summarize_wrist_errors <- function(data, metric_type = NULL, significant_only = FALSE) {
  data <-
    data |>
    mutate(
      p_wrist = p_wrist <= sig.level,
      wrist_error = if (significant_only) ifelse(p_wrist, wrist_error, NA_real_) else wrist_error
    )

  out <-
    data |>
    summarize(
      n = n(),
      wrist_median = median(abs(wrist_error), na.rm = TRUE),
      wrist_min = min(abs(wrist_error), na.rm = TRUE),
      wrist_max = max(abs(wrist_error), na.rm = TRUE),
      wrist_positive = sum(wrist_error > 0, na.rm = TRUE),
      wrist_negative = sum(wrist_error < 0, na.rm = TRUE),
      wrist_significant = sum(p_wrist, na.rm = TRUE),
      wrist_estimable = sum(!is.na(wrist_error))
    ) |>
    mutate(
      across(c(wrist_min, wrist_max), \(x) ifelse(is.infinite(x), NA_real_, x))
    )

  if (significant_only) {
    out <-
      out |>
      mutate(
        wrist_significant = NA_integer_,
        wrist_estimable = NA_integer_
      )
  }

  if (!is.null(metric_type)) {
    out <- mutate(out, metric_type = metric_type, .before = n)
  }

  out
}

summary_by_type <-
